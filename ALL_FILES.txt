const express = require('express');
const cors = require('cors');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = 3000;
const DB_PATH = path.join(__dirname, 'database.json');

// Middleware
app.use(cors());
app.use(express.json({ limit: '10mb' }));

// Serve static files from dist
app.use(express.static(path.join(__dirname, 'dist')));

// Database helpers
function readDB() {
  try {
    const data = fs.readFileSync(DB_PATH, 'utf-8');
    return JSON.parse(data);
  } catch (e) {
    return { records: [], version: '1.0', created: new Date().toISOString(), lastModified: '' };
  }
}

function writeDB(db) {
  db.lastModified = new Date().toISOString();
  if (!db.created) db.created = new Date().toISOString();
  fs.writeFileSync(DB_PATH, JSON.stringify(db, null, 2), 'utf-8');
}

// ========== API Routes ==========

// GET all records
app.get('/api/records', (req, res) => {
  const db = readDB();
  res.json(db.records);
});

// POST new record
app.post('/api/records', (req, res) => {
  const db = readDB();
  const record = req.body;
  db.records.push(record);
  writeDB(db);
  res.json({ success: true, id: record.id });
});

// PUT update record
app.put('/api/records/:id', (req, res) => {
  const db = readDB();
  const { id } = req.params;
  const idx = db.records.findIndex(r => r.id === id);
  if (idx === -1) {
    return res.status(404).json({ error: 'Запись не найдена' });
  }
  db.records[idx] = { ...db.records[idx], ...req.body };
  writeDB(db);
  res.json({ success: true });
});

// DELETE record
app.delete('/api/records/:id', (req, res) => {
  const db = readDB();
  const { id } = req.params;
  db.records = db.records.filter(r => r.id !== id);
  writeDB(db);
  res.json({ success: true });
});

// GET check duplicates
app.get('/api/check', (req, res) => {
  const db = readDB();
  const { meterNumber, modemType, simCardNumber, excludeId } = req.query;
  
  const result = {
    meterNumberExists: false,
    modemExists: false,
    simCardExists: false,
  };

  if (meterNumber) {
    result.meterNumberExists = db.records.some(
      r => r.meterNumber === meterNumber && r.id !== excludeId
    );
  }

  if (modemType && modemType !== 'встроенный') {
    result.modemExists = db.records.some(
      r => r.modemType === modemType && r.id !== excludeId
    );
  }

  if (simCardNumber) {
    result.simCardExists = db.records.some(
      r => r.simCardNumber === simCardNumber && r.id !== excludeId
    );
  }

  res.json(result);
});

// GET export all records (for xlsx export)
app.get('/api/export', (req, res) => {
  const db = readDB();
  res.json(db.records);
});

// Fallback to index.html for SPA
app.get('*', (req, res) => {
  res.sendFile(path.join(__dirname, 'dist', 'index.html'));
});

// Start server
app.listen(PORT, () => {
  console.log('');

import { MeterRecord } from './types';

const API_BASE = '/api';

// Cache for records
let recordsCache: MeterRecord[] | null = null;

export async function getRecords(): Promise<MeterRecord[]> {
  try {
    const response = await fetch(`${API_BASE}/records`);
    const data = await response.json();
    recordsCache = data;
    return data;
  } catch (e) {
    // Fallback to cache or empty
    return recordsCache || [];
  }
}

export async function addRecord(record: MeterRecord): Promise<void> {
  await fetch(`${API_BASE}/records`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(record),
  });
  recordsCache = null; // invalidate cache
}

export async function updateRecord(id: string, updates: Partial<MeterRecord>): Promise<void> {
  await fetch(`${API_BASE}/records/${id}`, {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(updates),
  });
  recordsCache = null;
}

export async function deleteRecord(id: string): Promise<void> {
  await fetch(`${API_BASE}/records/${id}`, {
    method: 'DELETE',
  });
  recordsCache = null;
}

export function generateId(): string {
  return Date.now().toString(36) + Math.random().toString(36).substr(2, 9);
}

export async function checkDuplicates(
  meterNumber?: string,
  modemType?: string,
  simCardNumber?: string,
  excludeId?: string
): Promise<{ meterNumberExists: boolean; modemExists: boolean; simCardExists: boolean }> {
  try {
    const params = new URLSearchParams();
    if (meterNumber) params.set('meterNumber', meterNumber);
    if (modemType) params.set('modemType', modemType);
    if (simCardNumber) params.set('simCardNumber', simCardNumber);
    if (excludeId) params.set('excludeId', excludeId);

    const response = await fetch(`${API_BASE}/check?${params.toString()}`);
    return await response.json();
  } catch (e) {
    return { meterNumberExists: false, modemExists: false, simCardExists: false };
  }
}

// Legacy sync functions for compatibility
export function getRecordsSync(): MeterRecord[] {
  return recordsCache || [];
}

export function setRecordsCache(records: MeterRecord[]): void {
  recordsCache = records;
}

    '№': idx + 1,
    'Код региона': r.regionCode,
    'Номер БС': r.bsNumber,
    'Тип действия': r.actionType,
    'Тип прибора учёта': r.meterType,

    'Тип модема': r.modemType,
    'Номер SIM-карты': r.simCardNumber,
    'Отметка': r.note,
    'Зарегистрировал': r.registeredBy,
    'Дата регистрации': r.registrationDate,
    'Адрес объекта': r.address,
    'Примечание': r.comment,
    'Статус': r.registrationStatus || '—',
    'Дата одобрения': r.statusDate || '',
    'Одобрил': r.statusAdmin || '',
  }));

  const ws = XLSX.utils.json_to_sheet(data);

  // Set column widths
  const colWidths = [
    { wch: 5 },   // №
    { wch: 10 },  // Код региона
    { wch: 12 },  // Номер БС
    { wch: 14 },  // Тип действия
    { wch: 20 },  // Тип прибора учёта
    { wch: 18 },  // Номер счётчика
    { wch: 20 },  // Тип модема
    { wch: 14 },  // Номер SIM-карты
    { wch: 15 },  // Отметка
    { wch: 15 },  // Зарегистрировал
    { wch: 20 },  // Дата регистрации
    { wch: 40 },  // Адрес объекта
    { wch: 30 },  // Примечание
    { wch: 10 },  // Статус
    { wch: 20 },  // Дата одобрения
    { wch: 15 },  // Одобрил
  ];
  ws['!cols'] = colWidths;

  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, 'Приборы учёта');

  const now = new Date();
  const dateStr = now.toLocaleDateString('ru-RU').replace(/\./g, '');
  const fileName = `Регистратор${dateStr}.xlsx`;

  XLSX.writeFile(wb, fileName);
}

import { useState, useEffect, useCallback, useRef } from 'react';
import { User, Page } from './types';
import Login from './components/Login';
import Layout from './components/Layout';
import ViewMeters from './components/ViewMeters';
import Registration from './components/Registration';
import Database from './components/Database';
import Feedback from './components/Feedback';
import Help from './components/Help';
import About from './components/About';

const SESSION_TIMEOUT = 10 * 60 * 1000; // 10 minutes

export default function App() {
  const [user, setUser] = useState<User | null>(null);
  const [currentPage, setCurrentPage] = useState<Page>('view');
  const sessionTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  const clearTimer = useCallback(() => {
    if (sessionTimerRef.current) {
      clearTimeout(sessionTimerRef.current);
      sessionTimerRef.current = null;
    }
  }, []);

  const startTimer = useCallback(() => {
    clearTimer();
    sessionTimerRef.current = setTimeout(() => {
      setUser(null);
      setCurrentPage('view');
    }, SESSION_TIMEOUT);
  }, [clearTimer]);

  const resetTimer = useCallback(() => {
    startTimer();
  }, [startTimer]);

  useEffect(() => {
    if (!user) {
      clearTimer();
      return;
    }

    const handleActivity = () => {
      resetTimer();
    };

    window.addEventListener('mousemove', handleActivity);
    window.addEventListener('keydown', handleActivity);
    window.addEventListener('click', handleActivity);
    window.addEventListener('scroll', handleActivity);

    startTimer();

    return () => {
      window.removeEventListener('mousemove', handleActivity);
      window.removeEventListener('keydown', handleActivity);
      window.removeEventListener('click', handleActivity);
      window.removeEventListener('scroll', handleActivity);
      clearTimer();
    };
  }, [user, startTimer, resetTimer, clearTimer]);

  const handleLogin = (loggedUser: User) => {
    setUser(loggedUser);
    setCurrentPage('view'); // Default page after login
  };

  const handleLogout = () => {
    setUser(null);
    setCurrentPage('view');
    clearTimer();
  };

  const handleNavigate = (page: Page) => {
    setCurrentPage(page);
    resetTimer();
  };

  if (!user) {
    return <Login onLogin={handleLogin} />;
  }

  const renderPage = () => {
    switch (currentPage) {
      case 'view':
        return <ViewMeters user={user} />;
      case 'registration':
        return <Registration user={user} />;
      case 'database':
        return <Database user={user} />;
      case 'feedback':
        return <Feedback />;
      case 'help':
        return <Help />;
      case 'about':
        return <About />;
      default:
        return <ViewMeters user={user} />;
    }
  };

  return (
    <Layout
      user={user}
      currentPage={currentPage}
      onNavigate={handleNavigate}
      onLogout={handleLogout}
    >
      {renderPage()}
    </Layout>
  );
}
